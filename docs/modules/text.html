Copy and paste

This is a complete HTML file. It needs no build step and no framework. The two scripts weigh 4.4 KB + 37.5 KB gzipped.

<!doctype html>
<html lang="en">
<head>
<meta charset="utf-8">
<meta name="viewport" content="width=device-width, initial-scale=1">
<title>Aurora Text (standalone)</title>
<style>
body { margin: 0; min-height: 100vh; display: grid; place-items: center; padding: 24px; background: #0b0b12; color: #ecebf5; font: 16px/1.5 system-ui, sans-serif; }
h2 { font-size: clamp(2rem, 6vw, 3.4rem); margin: 0; text-align: center; }
</style>
</head>
<body>

<h2 data-aurora-text="slide-in">Motion, unleashed.</h2>

JavaScript API

var fx = Aurora.text(document.querySelector('h2'), { effect: 'blur-reveal', split: 'words' });
fx.replay();               // play again
fx.update({ duration: 1200 }); // change options
fx.destroy();              // restore the original markup
